    el.style.left = Math.min(Math.max(0, cx - ox), maxX) + 'px';
    el.style.top  = Math.min(Math.max(0, cy - oy), maxY) + 'px';
  }

  function endDrag() { dragging = false; el.style.transition = ''; }

  el.addEventListener('mousedown',  e => { if (e.target.closest('#btn-reel-minimize')) return; startDrag(e.clientX, e.clientY); e.preventDefault(); });
  document.addEventListener('mousemove', e => moveDrag(e.clientX, e.clientY));
  document.addEventListener('mouseup',   endDrag);

  el.addEventListener('touchstart', e => { if (e.target.closest('#btn-reel-minimize')) return; const t = e.touches[0]; startDrag(t.clientX, t.clientY); }, { passive: true });
  document.addEventListener('touchmove',  e => { if (!dragging) return; const t = e.touches[0]; moveDrag(t.clientX, t.clientY); e.preventDefault(); }, { passive: false });
  document.addEventListener('touchend',   endDrag);

  el.addEventListener('click', e => { if (moved) { moved = false; e.stopImmediatePropagation(); e.preventDefault(); } });
})();
